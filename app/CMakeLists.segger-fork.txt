cmake_minimum_required(VERSION 3.20)
project(segger_poc C ASM)
set(TX_USER_FILE ${CMAKE_CURRENT_LIST_DIR}/tx_user_segger.h)
add_subdirectory(${THREADX_DIR} threadx)
target_compile_definitions(threadx PUBLIC TXPOC_QEMU ENABLE_TRACE_API=1 TXTRACE_BACKEND=0 TXTRACE_SEGGER=1)
target_include_directories(threadx PUBLIC ${SYSVIEW_DIR}/Config ${SYSVIEW_DIR}/SEGGER ${SYSVIEW_DIR}/SYSVIEW ${SYSVIEW_DIR}/Sample/ThreadX ${RTT_DIR}/Config ${RTT_DIR}/RTT)
add_library(sv STATIC ${SYSVIEW_DIR}/SYSVIEW/SEGGER_SYSVIEW.c ${SYSVIEW_DIR}/Sample/ThreadX/SEGGER_SYSVIEW_ThreadX.c
  ${SYSVIEW_DIR}/Sample/ThreadX/Config/Cortex-M/SEGGER_SYSVIEW_Config_ThreadX.c ${RTT_DIR}/RTT/SEGGER_RTT.c ${RTT_DIR}/RTT/SEGGER_RTT_ASM_ARMv7M.S)
target_link_libraries(sv PUBLIC threadx)
target_compile_definitions(sv PRIVATE "SEGGER_SYSVIEW_GET_TIMESTAMP()=txpoc_timestamp()")
target_link_libraries(threadx INTERFACE sv)
set(EX $ENV{HOME}/segger-threadx/ports/cortex_m4/gnu/example_build)  # upstream startup (fork example is stale)
set(EXLL ${THREADX_DIR}/ports/cortex_m4/gnu/example_build)
add_executable(demo ${APP} ${EX}/cortexm4_vectors.S ${EX}/cortexm4_crt0.S ${EXLL}/tx_initialize_low_level.S)
target_link_libraries(demo PRIVATE threadx)
target_link_options(demo PRIVATE -T${EX}/sample_threadx.ld -nostartfiles -Wl,--gc-sections)
